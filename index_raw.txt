        function saveRawData(content) {
            const request = indexedDB.open(DB_NAME, 1);
            request.onupgradeneeded = e => { e.target.result.createObjectStore(STORE_NAME); };
            request.onsuccess = e => {
                const db = e.target.result;
                const tx = db.transaction(STORE_NAME, "readwrite");
                tx.objectStore(STORE_NAME).put(content, "rawData");
            };
        }

        function loadRawData() {
            const request = indexedDB.open(DB_NAME, 1);
            request.onupgradeneeded = e => { e.target.result.createObjectStore(STORE_NAME); };
            request.onsuccess = e => {
                const db = e.target.result;
                // If the store exists, fetch. If not, maybe it was just created.
                if (db.objectStoreNames.contains(STORE_NAME)) {
                    const tx = db.transaction(STORE_NAME, "readonly");
                    const store = tx.objectStore(STORE_NAME);
                    const req = store.get("rawData");
                    req.onsuccess = () => {
                        if (req.result) {
                            document.getElementById('inputData').value = req.result;
                            processData();
                        }
                    }
                }
            };
        }

        function processLine(line) {
            const cleanLine = line.replace(/(^"|"$)/g, '').replace(/""/g, '"');
            let parts = cleanLine.split('\t');
            if (parts.length < 5) parts = cleanLine.split(';');
            if (parts.length < 5) return null;

            const isSiape = cleanLine.toUpperCase().includes('SIAPE');

            if (isSiape) {
                let matricula = parts[0] ? parts[0].trim() : "";
                let rawCpf = (parts[1] || "").replace(/\D/g, '').replace(/^0+/, '');
                let cpfFormatado = rawCpf;
                if (rawCpf.length > 2) {
                    cpfFormatado = rawCpf.slice(0, -2) + "-" + rawCpf.slice(-2);
                }

                let nome = parts[2] ? parts[2].trim() : "";
                let nasc = parts[4] ? parts[4].trim() : "";
                let cargo = parts[7] ? parts[7].trim() : "";
                let valorCru = parts[9] ? parts[9].trim() : "";

                let banco = parts[13] ? parts[13].trim() : "";
                let agencia = parts[14] ? parts[14].trim() : "";
                let conta = parts[15] ? parts[15].trim() : "";

                let uf = parts[16] ? parts[16].trim() : "";
                let cidade = parts[17] ? parts[17].trim() : "";
                let bairro = parts[18] ? parts[18].trim() : "";
                let rua = parts[19] ? parts[19].trim() : "";
                let cep = parts[20] ? parts[20].trim() : "";

                let rawPhones = [];
                if (parts[21]) rawPhones.push(parts[21].replace(/\D/g, ''));
                if (parts[22]) rawPhones.push(parts[22].replace(/\D/g, ''));
                if (parts[23]) rawPhones.push(parts[23].replace(/\D/g, ''));
                let phones = [...new Set(rawPhones.filter(p => p.length >= 8))];

                return {
                    isSiape: true,
                    nome: nome,
                    cpf: cpfFormatado,
                    nb: matricula,
                    nascimento: nasc,
                    cargo: cargo,
                    valor: valorCru ? "R$ " + valorCru : "",
                    banco: banco,
                    agencia: agencia,
                    conta: conta,
                    ufRaw: uf,
                    cidade: cidade,
                    bairro: bairro,
                    rua: rua,
                    cep: cep,
                    telefones: phones,
                    valorRaw: parseFloat(valorCru.replace(',', '.')) || 0
                };
            } else {
                let nb = parts[0] ? parts[0].trim() : "";
                let rawCpf = parts[1] || "";
                let cpfDigits = rawCpf.replace(/\D/g, '');
                let cpfFormatado = cpfDigits;
                if (cpfDigits.length === 11) {
                    cpfFormatado = cpfDigits.replace(/(\d{3})(\d{3})(\d{3})(\d{2})/, "$1.$2.$3-$4");
                }

                let nome = parts[2] ? parts[2].trim() : "";
                let valorCru = parts[6] ? parts[6].trim() : "";
                let nasc = parts[7] ? parts[7].trim() : "";
                let especie = parts[10] ? parts[10].trim() : "";
                let banco = parts[12] ? parts[12].replace(/^0+/, '').trim() : "";
                let agencia = parts[13] ? parts[13].trim() : "";
                let conta = parts[14] ? parts[14].trim() : "";
                let municipio = parts[15] ? parts[15].trim() : "";
                let uf = parts[16] ? parts[16].trim() : "";
                let bairro = parts[17] ? parts[17].trim() : "";
                let rua = parts[18] ? parts[18].trim() : "";

                let enderecoCompleto = "";
                if (rua && rua !== "NAO" && rua !== "") {
                    enderecoCompleto += rua;
                    if (bairro && bairro !== "NAO" && bairro !== "") enderecoCompleto += ", " + bairro;
                    if (municipio && municipio !== "NAO" && municipio !== "") enderecoCompleto += " - " + municipio;
                    if (uf && uf !== "NAO" && uf !== "") enderecoCompleto += " " + uf;
                }

                let rawPhones = [];
                if (parts[29]) rawPhones.push(parts[29].replace(/\D/g, ''));
                if (parts[30]) rawPhones.push(parts[30].replace(/\D/g, ''));
                if (parts[31]) rawPhones.push(parts[31].replace(/\D/g, ''));
                let phones = [...new Set(rawPhones.filter(p => p.length >= 8))];

                let rawEmails = [];
                if (parts[32] && parts[32].includes("@")) rawEmails.push(parts[32].trim());
                if (parts[33] && parts[33].includes("@")) rawEmails.push(parts[33].trim());
                if (parts[34] && parts[34].includes("@")) rawEmails.push(parts[34].trim());

                return {
                    isSiape: false,
                    nb: nb,
                    cpf: cpfFormatado,
                    nome: nome,
                    valor: valorCru ? "R$ " + valorCru : "",

                function generateFinalText(data) {
            let finalText = "";
            if (data.isSiape) {
                if (data.nome) finalText += `?? **NOME:** ${data.nome}\n`;
                if (data.cpf) finalText += `?? **CPF:** ${data.cpf}\n`;
                if (data.nascimento) finalText += `?? **NASCIMENTO:** ${data.nascimento}\n`;
                if (data.cargo) finalText += `CARGO: ${data.cargo}\n`;
                if (data.valor) finalText += `?? **VALOR SALARIO:** ${data.valor}\n`;
                
                finalText += `?? **BANCO:** ${data.banco || "-"} | **AG:** ${data.agencia || "-"} | **CC:** ${data.conta || "-"}\n`;
                
                let end = `?? **ENDEREÇO:** `;
                if (data.bairro) end += `${data.bairro}, `;
                if (data.cidade) end += `"${data.cidade}" `;
                if (data.rua) end += `${data.rua} `;
                end += `- (${data.cep ? data.cep : "em branco pois não tem cep"}) `;
                if (data.ufRaw) end += `${data.ufRaw}`;
                finalText += end + `\n`;

                if (data.telefones && data.telefones.length > 0) {
                    finalText += `TELEFONES:\n\n`;
                    data.telefones.forEach(p => finalText += `${p}\n`);
                }
            } else {
                if (data.nome) finalText += `?? **NOME:** ${data.nome}\n`;
                if (data.cpf) finalText += `?? **CPF:** ${data.cpf}\n`;
                if (data.nascimento) finalText += `?? **NASCIMENTO:** ${data.nascimento}\n`;
                if (data.valor) finalText += `?? **VALOR BENEFÍCIO:** ${data.valor}\n`;
                if (data.banco || data.agencia || data.conta) {
                    finalText += `?? **BANCO:** ${data.banco || "-"} | **AG:** ${data.agencia || "-"} | **CC:** ${data.conta || "-"}\n`;
                }
                if (data.especie) finalText += `?? **ESPÉCIE:** ${data.especie}\n`;
                if (data.endereco) finalText += `?? **ENDEREÇO:** ${data.endereco}\n\n`;
                if (data.emails && data.emails.length > 0) {
                    finalText += `?? **EMAILS:**\n`;
                    data.emails.forEach(e => finalText += `  • ${e}\n`);
                    finalText += `\n`;
                }
                if (data.telefones && data.telefones.length > 0) {
                    finalText += `?? **TELEFONES:**\n`;
                    data.telefones.forEach(p => finalText += `  • ${p}\n`);
                }
            }
            return finalText.trim() + "\n";
        }

        function processData() {

                // Configura Dashboard Iniciais

        function applyFilters() {
            // Remove acentos da string de busca para facilitar
            const queryRaw = document.getElementById('filterBusca').value.toLowerCase();
            const query = queryRaw.normalize("NFD").replace(/[\u0300-\u036f]/g, "");

            const fonte = document.getElementById('filterFonte').value;
            const uf = document.getElementById('filterUF').value.toUpperCase();
            const banco = document.getElementById('filterBanco').value;
            const rendaMin = parseFloat(document.getElementById('filterRendaMin').value) || 0;
            const rendaMax = parseFloat(document.getElementById('filterRendaMax').value) || Infinity;

            window.filteredDataArray = window.processedDataArray.filter(data => {
                // Filtro Busca (Nome ou CPF)
                const nomeNormalizado = (data.nome || "").toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "");

                const searchMatch = nomeNormalizado.includes(query) ||
                    (data.cpf && data.cpf.replace(/\D/g, '').includes(query.replace(/\D/g, '')));

                // Filtro Fonte (INSS vs SIAPE)
                const fonteMatch = fonte === "" ||
                    (fonte === "SIAPE" && data.isSiape) ||
                    (fonte === "INSS" && !data.isSiape);

                // Filtro UF
                const ufMatch = uf === "" || (data.ufRaw || "").toUpperCase() === uf;

                // Filtro Banco
                const bancoMatch = banco === "" || (data.banco || "").trim() === banco;

                // Filtro Renda
                const rendaMatch = data.valorRaw >= rendaMin && data.valorRaw <= rendaMax;

                return searchMatch && fonteMatch && ufMatch && bancoMatch && rendaMatch;
            });

            updateDashboardStats();
            currentPage = 1;
            renderPage(currentPage);
        }

                }
            });

            document.getElementById('statTotal').innerText = total;
            document.getElementById('statContatos').innerText = comContato;
        }

        function copyWhatsAppMessage(index, btnElem) {
            const data = window.processedDataArray[index];
            if (!data) return;

            // Pega o template do Textarea definido pelo usuÃ¡rio
            let template = document.getElementById('whatsappTemplate').value;

            // FormataÃ§Ã£o de blocos extra (emails e telefones)
            let emailBlock = "";
            if (data.emails && data.emails.length > 0) {
                emailBlock = `ðŸ“§ *EMAILS:* ${data.emails.join(", ")}`;
            }

            let phoneBlock = "";
            if (data.telefones && data.telefones.length > 0) {
                const uniquePhones = [...new Set(data.telefones)];
                phoneBlock = `ðŸ“± *TELEFONES:* ${uniquePhones.join(", ")}`;
            }

            // Substituir variÃ¡veis ou deletar as linhas que ficariam penduradas (vazias) sem as variÃ¡veis
            let message = template;

            const mapReplace = {
                "{nome}": data.nome || "",
                "{cpf}": data.cpf || "",
                "{nb}": data.nb || "",
                "{valor}": data.valor || "",
                "{endereco}": data.endereco || ""
            };

            for (const [key, value] of Object.entries(mapReplace)) {
                if (!value) {
                    // Se a variÃ¡vel Ã© vazia, regex para apagar a linha inteira em que a TAG original estava
                    const emptyRegex = new RegExp(`^.*${key}.*$\\n?`, 'gm');
                    message = message.replace(emptyRegex, '');
                } else {
                    message = message.replace(new RegExp(key, 'g'), value);
                }
            }

            // Emails e Telefones (blocos em vez de in-line)
            if (!emailBlock) {
                message = message.replace(/^.*{emails}.*$\n?/gm, '');
            } else {
                message = message.replace(/{emails}/g, emailBlock);
            }

            if (!phoneBlock) {
                message = message.replace(/^.*{telefones}.*$\n?/gm, '');
            } else {
                message = message.replace(/{telefones}/g, phoneBlock);
            }

            navigator.clipboard.writeText(message).then(() => {
                const originalContent = btnElem.innerHTML;
                btnElem.innerHTML = `<span>Mensagem Copiada!</span>`;
                btnElem.style.background = "#128C7E";
                setTimeout(() => {
                    btnElem.innerHTML = originalContent;
                    btnElem.style.background = "#25D366";
                }, 2000);
            });
        }

        async function generatePDFWorker(data, isPreview) {
            const { jsPDF } = window.jspdf;
            const doc = new jsPDF();

            const loadImage = (url) => {
                return new Promise((resolve, reject) => {
                    const img = new Image();
                    img.crossOrigin = "Anonymous";
                    img.src = url;
                    img.onload = () => {
                        const canvas = document.createElement("canvas");
                        canvas.width = img.width;
                        canvas.height = img.height;
                        const ctx = canvas.getContext("2d");
                        ctx.drawImage(img, 0, 0);
                        resolve(canvas.toDataURL("image/png"));
                    };
                    img.onerror = (e) => {
                        console.error("Error loading image", url, e);
                        resolve(null);
                    };
                });
            };

            const brasaoUrl = document.getElementById('pdfLogoTop').value;
            const bbLogoUrl = document.getElementById('pdfLogoBottom').value;

            const [brasaoBase64, bbLogoBase64] = await Promise.all([
                loadImage(brasaoUrl),
                loadImage(bbLogoUrl)
            ]);

            if (!brasaoBase64 && !bbLogoBase64) {
                console.warn("Uma ou mais imagens falharam em carregar para o PDF. Elas podem ser muito grandes ou estar bloqueadas (CORS).");
            }

            const pageWidth = doc.internal.pageSize.getWidth();
            const pageHeight = doc.internal.pageSize.getHeight();
            const margin = 20;

            let yPos = 20;

            // 1. Logo Topo
            if (brasaoBase64) {
                const imgWidth = Number(document.getElementById('topW').value) || 30;
                const imgHeight = Number(document.getElementById('topH').value) || 30;
                const x = (pageWidth - imgWidth) / 2;
                doc.addImage(brasaoBase64, 'PNG', x, yPos, imgWidth, imgHeight);
                yPos += imgHeight + 15;
            } else {
                yPos += 20;
            }

            // 2. TÃ­tulo do Documento Customizado
            doc.setFontSize(16);
            doc.setFont("helvetica", "bold");
            const title = document.getElementById('pdfTitle').value || "DOCUMENTO OFICIAL";
            const titleSplit = doc.splitTextToSize(title, pageWidth - 2 * margin);
            doc.text(titleSplit, pageWidth / 2, yPos, { align: 'center' });
            yPos += (titleSplit.length * 7) + 10;

            // GeraÃ§Ã£o de Datas
            const todayObj = new Date();
            const yesterdayObj = new Date(todayObj);
            yesterdayObj.setDate(todayObj.getDate() - 1);

            const fmtDate = (d) => `${String(d.getDate()).padStart(2, '0')}/${String(d.getMonth() + 1).padStart(2, '0')}/${d.getFullYear()}`;
            const dataHoje = fmtDate(todayObj);
            const dataOntem = fmtDate(yesterdayObj);

            // 3. RenderizaÃ§Ã£o Multi-Linhas
            let rawPdfText = document.getElementById('pdfTemplate').value || "";

            const pdfMapReplace = {
                "{nome}": data.nome || "-",
                "{cpf}": data.cpf || "-",
                "{nb}": data.nb || "-",
                "{valor}": data.valor || "-",
                "{endereco}": data.endereco || "-",
                "{agencia}": data.agencia || "-",
                "{conta}": data.conta || "-",
                "{data_hoje}": dataHoje,
                "{data_ontem}": dataOntem
            };

            for (const [key, value] of Object.entries(pdfMapReplace)) {
                rawPdfText = rawPdfText.replace(new RegExp(key, 'g'), value);
            }

            doc.setFontSize(12);

            const lines = rawPdfText.split('\n');
            for (let line of lines) {
                line = line.replace('\r', '');
                if (line.trim() === '') {
                    yPos += 7;
                    continue;
                }

                if (line.startsWith('*') && line.endsWith('*')) {
                    doc.setFont("helvetica", "bold");
                    line = line.substring(1, line.length - 1);
                } else {
                    doc.setFont("helvetica", "normal");
                }

                // ProteÃ§Ã£o de overflow de pÃ¡gina
                const splitLine = doc.splitTextToSize(line, pageWidth - 2 * margin);
                if (yPos + (splitLine.length * 7) > pageHeight - 40) {
                    doc.addPage();
                    yPos = 20;
                }
                doc.text(splitLine, margin, yPos);
                yPos += splitLine.length * 7;
            }

            // Footer: Logotipo Base
            if (bbLogoBase64) {
                const botWidth = Number(document.getElementById('botW').value) || 40;
                const botHeight = Number(document.getElementById('botH').value) || 10;
                const xBB = (pageWidth - botWidth) / 2;
                const yBB = pageHeight - botHeight - 20;
                doc.addImage(bbLogoBase64, 'PNG', xBB, yBB, botWidth, botHeight);
            }

            // Export logic (Preview vs Download)
            if (isPreview) {
                const pdfDataUri = doc.output('datauristring');
                openPreviewModal(pdfDataUri);
            } else {
                const filename = `${data.nome.replace(/\s+/g, '_')}_BLOQUEIO.pdf`;
                doc.save(filename);
            }
        }
